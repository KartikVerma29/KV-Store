   }

#include "FileWAL.h"
#include <cstdint>
#include <unistd.h>

FileOffset FileWAL::append(const LogEntry& log){
   std::vector<uint8_t> bytes = log.encode();
   std::lock_guard<std::mutex> lock(writeMutex);
   fseek(fileHandle, 0, SEEK_END);
   uint64_t offset = ftell(fileHandle);
   fwrite(bytes.data(), 1, bytes.size(), fileHandle);
   return FileOffset{offset, bytes.size()};
}

std::vector<LogEntry> FileWAL::replay(){
   fseek(fileHandle, 0, SEEK_SET);
   std::vector<LogEntry> logs;
   uint32_t payloadSize;

   while(std::fread(&payloadSize, sizeof(uint32_t),1, fileHandle)==1 ){
      uint32_t totalRead = sizeof(uint32_t)*2 + payloadSize;
      std::vector<uint8_t> buffer(totalRead);
      std::memcpy(buffer.data(), &payloadSize, sizeof(uint32_t));

      std::fread(buffer.data() + sizeof(uint32_t), 1, totalRead - sizeof(uint32_t), fileHandle);
      logs.push_back(LogEntry::decode(buffer));
   }
   return logs;
}

void FileWAL::sync(){
   fflush(fileHandle);
   fsync(fileno(fileHandle));
}

FileWAL::~FileWAL(){
   std::fclose(fileHandle);
}

#pragma once
#include <cstdint>
#include <string>
#include <vector>
#include <chrono>
#include <cstring>

class Record{
private:
   std::string key;
   std::string value;
   bool tombstone;
   uint64_t timestamp;

   Record(std::string k, std::string v, bool tomb, uint64_t ts)
      :key(std::move(k)), value(std::move(v)), tombstone(tomb), timestamp(ts)
   {}

public:

   Record(const std::string& key, const std::string& value):key(key), value(value), tombstone(false){
      timestamp = std::chrono::duration_cast<std::chrono::milliseconds>(std::chrono::system_clock::now().time_since_epoch()).count();
   }

   const std::string& getKey() const{
      return key;
   }

   bool isTomb() const{
      return tombstone;
   }

   std::vector<uint8_t> serialize() const{
      std::vector<uint8_t> buffer(size());
      size_t offset=0;
      auto write = [&](const void* data, size_t len){
         std::memcpy(buffer.data()+offset, data, len);
         offset+=len;
      };

      size_t kLen = key.length();
      write(&kLen, sizeof(kLen));
      write(key.data(), kLen);

      size_t vLen = value.length();
      write(&vLen, sizeof(vLen));
      write(value.data(), vLen);

      write(& tombstone, sizeof(tombstone));
      write(& timestamp, sizeof(timestamp));
      return buffer;
   }

   Record static deserialize(const std::vector<uint8_t>& buffer){
      size_t offset = 0;

      auto read = [&](void* dest, size_t len){
         std::memcpy(dest, buffer.data()+offset,len);
         offset+=len;
      };

      size_t kLen;
      read(&kLen, sizeof(kLen));
      std::string k(reinterpret_cast<const char*>(buffer.data()+offset), kLen);
      offset+=kLen;

      size_t vLen;
      read(&vLen, sizeof(vLen));
      std::string v(reinterpret_cast<const char*>(buffer.data()+offset), vLen);
      offset+=vLen;

      bool tomb; read(&tomb, sizeof(tomb));

      uint64_t ts; read(&ts, sizeof(ts));
      
      return Record(std::move(k), std::move(v), tomb, ts);
   }

   size_t size() const{
      return sizeof(bool) + sizeof(uint64_t) + (sizeof(size_t)*2) + key.length() + value.length();
   };

};

#pragma once
#include "Record.h"
#include <cstddef>
#include <cstdint>
#include <cstring>
#include <vector>
#include "../../utils/CRC32.h"

class LogEntry{
private:
   Record record;
   uint32_t checksum; //CRC32

   LogEntry(Record r, uint32_t c): record(std::move(r)), checksum(c) {}

public:
   LogEntry(const Record& record): record(record){
      std::vector<uint8_t> bytes = record.serialize();
      checksum = utils::CRC32(bytes.data(), bytes.size());
   }

   const Record& getRecord() const{
      return record;
   }

   const uint32_t getcs() const{
      return checksum;
   }

   std::vector<uint8_t> encode() const{
      std::vector<uint8_t> bytes = record.serialize();
      uint32_t payloadSize = bytes.size();

      std::vector<uint8_t> buffer(sizeof(uint32_t)*2+payloadSize);

      std::memcpy(buffer.data(), &payloadSize, sizeof(uint32_t));
      std::memcpy(buffer.data()+sizeof(uint32_t), &checksum, sizeof(uint32_t));
      std::memcpy(buffer.data()+(sizeof(uint32_t)*2), bytes.data(), payloadSize);
      return buffer;
   }

   LogEntry static decode(const std::vector<uint8_t>& buffer) {
      uint32_t c;
      std::memcpy(&c, buffer.data()+4, sizeof(uint32_t));
      std::vector<uint8_t> bytes(buffer.begin()+8, buffer.end());
      return LogEntry(Record::deserialize(bytes),c);
   }

   bool validateChecksum() const{
      std::vector<uint8_t> bytes = record.serialize();
      uint32_t calculated  = utils::CRC32(bytes.data(), bytes.size());
      return checksum==calculated;
   }

};

#pragma once

#include <cstddef>
#include <cstdint>
#include <array>

namespace utils
{
   constexpr std::array<uint32_t, 256> generate_table(){
      std::array<uint32_t, 256> table{};
      uint32_t poly = 0xEDB88320;
      for(uint32_t i=0; i<256; i++){
         uint32_t rem=i;
         for(int j=0; j<8; j++){
            if(rem&1) rem = (rem>>1)^poly;
            else rem>>=1;
         }
         table[i]=rem;
      }
      return table;
   }

   constexpr auto crc32Table = generate_table();

   inline uint32_t CRC32(const uint8_t* data, size_t size){
      uint32_t rem = 0xFFFFFFFF;
      for(size_t i=0; i<size; i++){
         uint8_t byte = data[i];
         rem=(rem>>8)^crc32Table[(rem^byte)&0xFF];
      }
      return rem^0XFFFFFFF;
   }

}
